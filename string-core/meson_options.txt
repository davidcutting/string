# Window System Integration backend.
#   sdl     - default; SDL3 backend (from the sdl3 wrap)
#   wayland - from-scratch Wayland client (EXPERIMENTAL, Linux-only, needs protocol codegen)
#   glfw    - source kept in-tree but not wired into the wrap-based build
option('wsi', type: 'combo', choices: ['sdl', 'wayland', 'glfw'], value: 'sdl',
       description: 'Windowing / input backend')

# Build the unit/integration tests (requires the gtest wrap).
option('tests', type: 'boolean', value: false,
       description: 'Build the test suite')

# Compile in the Tracy profiler client (links the tracy wrap, defines STRING_PROFILE).
# The profiler viewer is a separate tool (nixpkgs `tracy` in the dev shell).
option('tracy', type: 'boolean', value: false,
       description: 'Enable the Tracy profiler client')

# Link the Slang shader compiler.
#   enabled  - default; shaders compile at runtime, hot-reload and the Shadertoy REPL work
#   disabled - SHIP MODE: the Slang path is compiled out entirely and shaders are served only from
#              the prebuilt program cache. Drops ~33MB of DLLs from a Windows package. A cache miss
#              becomes a hard error, so the package MUST carry a cache warmed by an enabled build
#              (see docs/off-nix-build.md).
option('slang', type: 'feature', value: 'enabled',
       description: 'Link Slang for runtime shader compilation (disable for a cache-only ship build)')
